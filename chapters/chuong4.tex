\subsection{4.1. Xác định hệ quản trị cơ sở dữ liệu}
\label{sec:he_quan_tri_csdl}

Theo yêu cầu phi chức năng đã xác định ở Chương 1 (mục 1.d --- Non-Functional Requirements), hệ thống FamilyConnect được yêu cầu sử dụng PostgreSQL làm hệ quản trị cơ sở dữ liệu (DBMS). Mục này trình bày các tiêu chí kỹ thuật củng cố cho lựa chọn đó, đối chiếu với các phương án thay thế phổ biến.

Các tiêu chí lựa chọn được xem xét gồm:

\begin{itemize}
	\item Tính toàn vẹn dữ liệu và tuân thủ ACID: dữ liệu gia phả, hôn nhân, quan hệ huyết thống đòi hỏi ràng buộc tham chiếu (FK) và ràng buộc miền giá trị (CHECK) chặt chẽ --- phù hợp với mô hình quan hệ hơn là NoSQL.
	\item Hỗ trợ truy vấn đệ quy (Recursive CTE): cây gia phả và cây phân cấp chi/nhánh (Branch tự tham chiếu) cần truy vấn đệ quy nhiều tầng, được PostgreSQL hỗ trợ tốt qua cú pháp WITH RECURSIVE.
	\item Khả năng mở rộng (extensibility): PostgreSQL hỗ trợ các extension như pgcrypto (sinh khóa UUID), pg\_trgm (tìm kiếm gần đúng) và tsvector/tsquery (tìm kiếm toàn văn) --- phục vụ trực tiếp các tính năng AI-assisted Search và Family Directory.
	\item Chi phí và giấy phép: PostgreSQL là phần mềm mã nguồn mở, miễn phí bản quyền, phù hợp với quy mô đồ án và khả năng triển khai thực tế.
	\item Khả năng triển khai theo container: PostgreSQL có image chính thức trên Docker Hub, phù hợp với yêu cầu phi chức năng "Docker Deployment" đã đặt ra ở Chương 1.
	\item Hệ sinh thái và khả năng tích hợp: PostgreSQL tương thích tốt với các driver phổ biến (Node.js, Python, Java) để kết nối với RESTful API Layer của hệ thống.
\end{itemize}

\subsection{4.2. Tạo Index tối ưu truy vấn}
\label{sec:tao_index}

PostgreSQL chỉ tự động tạo chỉ mục (index) cho khóa chính (PRIMARY KEY) và các cột có ràng buộc UNIQUE; các cột khóa ngoại (FK) và các cột thường xuyên xuất hiện trong mệnh đề WHERE/JOIN/ORDER BY không được tự động đánh chỉ mục. Do đó, cần chủ động tạo thêm chỉ mục cho các nhóm cột sau:

\begin{itemize}
	\item Cột khóa ngoại (FK): tăng tốc các phép JOIN và ràng buộc tham chiếu khi xóa/cập nhật bản ghi cha.
	\item Cột/nhóm cột thường dùng để lọc và sắp xếp trong các truy vấn nghiệp vụ lặp lại (ví dụ: family\_id kết hợp created\_at để lấy bảng tin mới nhất theo dòng họ).
	\item Cột phục vụ tra cứu danh bạ gia đình (nghề nghiệp, nơi ở, thế hệ).
	\item Chỉ mục đặc biệt GIN kết hợp pg\_trgm cho tìm kiếm gần đúng theo họ tên, và GIN kết hợp tsvector cho tìm kiếm toàn văn nội dung bài viết/câu chuyện gia đình.
\end{itemize}

\subsection{4.3. Giả lập các câu truy vấn để giải quyết vấn đề}
\label{sec:gia_lap_truy_van}

Phần này minh họa một số câu truy vấn SQL tiêu biểu, mô phỏng các bài toán nghiệp vụ thực tế của FamilyConnect và chỉ ra chỉ mục tương ứng (mục 4.2) được tận dụng để tối ưu hiệu năng.

\textbf{Truy vấn 1: Hiển thị cây gia phả của một dòng họ kèm quan hệ huyết thống}

\begin{lstlisting}
	SELECT fm.member_id, fm.full_name, fm.birth_date, fm.death_date,
	r.relation_type, pr.full_name AS quan_he_voi
	FROM family_members fm
	LEFT JOIN relationships r ON r.member_id_a = fm.member_id
	LEFT JOIN family_members pr ON pr.member_id = r.member_id_b
	WHERE fm.family_id = :family_id
	ORDER BY fm.generation_level, fm.full_name;
\end{lstlisting}

\textit{Chỉ mục idx\_members\_family và idx\_rel\_member\_a tăng tốc thao tác lọc theo dòng họ và JOIN với bảng quan hệ huyết thống.}

\textbf{Truy vấn 2: Tìm kiếm thành viên theo nghề nghiệp và nơi ở (Family Directory)}

\begin{lstlisting}
	SELECT member_id, full_name, occupation, location, generation_level
	FROM family_members
	WHERE family_id = :family_id
	AND occupation ILIKE '%ky su%'
	AND location = 'Ho Chi Minh';
\end{lstlisting}

\textit{Tận dụng idx\_members\_occupation và idx\_members\_location để giới hạn nhanh tập kết quả trước khi lọc chi tiết (ví dụ minh họa: occupation chứa "kỹ sư", location = "Hồ Chí Minh").}

\textbf{Truy vấn 3: Lấy 10 bài viết mới nhất của một dòng họ kèm số lượng bình luận và reaction}

\begin{lstlisting}
	SELECT p.post_id, p.content, p.created_at,
	COUNT(DISTINCT c.comment_id) AS so_binh_luan,
	COUNT(DISTINCT r.reaction_id) AS so_reaction
	FROM posts p
	LEFT JOIN comments c ON c.post_id = p.post_id
	LEFT JOIN reactions r ON r.post_id = p.post_id
	WHERE p.family_id = :family_id
	GROUP BY p.post_id
	ORDER BY p.created_at DESC
	LIMIT 10;
\end{lstlisting}

\textit{Chỉ mục composite idx\_posts\_family\_created (family\_id, created\_at DESC) cho phép lọc và sắp xếp chỉ bằng một lần quét chỉ mục (index scan), tránh phải sort toàn bảng.}

\textbf{Truy vấn 4: Danh sách sự kiện trong 30 ngày tới kèm số người xác nhận tham gia}

\begin{lstlisting}
	SELECT e.event_id, e.title, e.start_time,
	COUNT(ep.user_id) FILTER (WHERE ep.rsvp_status = 'yes')
	AS so_nguoi_xac_nhan
	FROM events e
	LEFT JOIN event_participants ep ON ep.event_id = e.event_id
	WHERE e.family_id = :family_id
	AND e.start_time BETWEEN CURRENT_TIMESTAMP
	AND CURRENT_TIMESTAMP + INTERVAL '30 days'
	GROUP BY e.event_id
	ORDER BY e.start_time;
\end{lstlisting}

\textit{Chỉ mục idx\_events\_family\_start (family\_id, start\_time) tối ưu cả điều kiện lọc theo dòng họ lẫn khoảng thời gian.}

\textbf{Truy vấn 5: Thống kê số lượng thành viên theo từng thế hệ (Dashboard -- Demographics)}

\begin{lstlisting}
	SELECT generation_level, COUNT(*) AS so_thanh_vien
	FROM family_members
	WHERE family_id = :family_id
	GROUP BY generation_level
	ORDER BY generation_level;
\end{lstlisting}

\textit{Chỉ mục idx\_members\_generation (family\_id, generation\_level) cho phép PostgreSQL tính GROUP BY trực tiếp trên chỉ mục (index-only scan) mà không cần truy cập bảng dữ liệu gốc.}

\textbf{Truy vấn 6: Lấy 20 lượt truy vấn AI gần nhất của một người dùng}

\begin{lstlisting}
	SELECT log_id, query_text, query_type, created_at
	FROM ai_query_logs
	WHERE user_id = :user_id
	ORDER BY created_at DESC
	LIMIT 20;
\end{lstlisting}

\textit{Chỉ mục idx\_query\_user\_time (user\_id, created\_at DESC) tránh phải sắp xếp lại toàn bộ kết quả sau khi lọc.}

\textbf{Truy vấn 7: Tra cứu nhật ký kiểm toán (audit) liên quan đến một bản ghi cụ thể}

\begin{lstlisting}
	SELECT log_id, user_id, action, created_at
	FROM audit_logs
	WHERE target_table = 'family_members'
	AND target_id = :member_id
	ORDER BY created_at DESC;
\end{lstlisting}

\textit{Chỉ mục idx\_audit\_table\_target (target\_table, target\_id) tăng tốc tra cứu lịch sử thay đổi phục vụ kiểm duyệt và điều tra sự cố.}

\textbf{Truy vấn 8: Tìm kiếm gần đúng theo họ tên thành viên (hỗ trợ gõ sai/không dấu)}

\begin{lstlisting}
	SELECT member_id, full_name,
	similarity(full_name, :keyword) AS do_tuong_dong
	FROM family_members
	WHERE full_name % :keyword
	ORDER BY do_tuong_dong DESC
	LIMIT 10;
\end{lstlisting}

\textit{Chỉ mục GIN idx\_members\_fullname\_trgm (pg\_trgm) cho phép so khớp gần đúng thay vì phải so khớp chính xác chuỗi ký tự.}

\textbf{Truy vấn 9: Tìm kiếm toàn văn nội dung bài viết theo từ khóa}

\begin{lstlisting}
	SELECT post_id, content
	FROM posts
	WHERE to_tsvector('simple', content) @@ plainto_tsquery('simple', :keyword);
\end{lstlisting}

\textit{Chỉ mục GIN idx\_posts\_content\_fts giúp thao tác tìm kiếm từ khóa trong nội dung văn bản dài đạt hiệu năng gần O(log n) thay vì quét tuần tự toàn bảng.}

\textbf{Truy vấn 10: Sinh số liệu báo cáo thống kê bài viết và sự kiện theo tháng cho một dòng họ}

\begin{lstlisting}
	SELECT date_trunc('month', COALESCE(p.created_at, e.start_time)) AS thang,
	COUNT(DISTINCT p.post_id) AS so_bai_viet,
	COUNT(DISTINCT e.event_id) AS so_su_kien
	FROM families f
	LEFT JOIN posts p ON p.family_id = f.family_id
	LEFT JOIN events e ON e.family_id = f.family_id
	WHERE f.family_id = :family_id
	GROUP BY thang
	ORDER BY thang;
\end{lstlisting}

\textit{Kết quả truy vấn này là số liệu nguồn được lưu lại thành một bản ghi trong bảng REPORT (report\_type = 'monthly\_activity') mỗi khi người dùng yêu cầu xuất báo cáo.}
