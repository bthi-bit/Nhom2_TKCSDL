{\LARGE\bfseries PHỤ LỤC A: SCRIPT T-SQL TẠO CƠ Sở Dữ LIỆU FAMILYCONNECTDB}\par\vspace{1em}

\emph{Toàn bộ script tạo CSDL FamilyConnectDB (33 quan hệ), được sinh trực tiếp từ lược đồ quan hệ đã chuẩn hóa ở Chương 3 và bảng ánh xạ kiểu dữ liệu logic sang vật lý ở Mục 4.2. Đây là nội dung file schema\_sqlserver.sql được nhắc đến xuyên suốt đồ án (Mục 1.3, 3.3, 4.4).}

\begin{lstlisting}[language=SQL]
CREATE DATABASE FamilyConnectDB;
GO
USE FamilyConnectDB;
GO
 
-- 1. Users
CREATE TABLE Users (
    UserId          UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    Username        VARCHAR(50) NOT NULL UNIQUE,
    Email           VARCHAR(100) NOT NULL UNIQUE,
    Phone           VARCHAR(20) NULL,
    PasswordHash    VARCHAR(255) NOT NULL,
    FullName        NVARCHAR(100) NOT NULL,
    AvatarUrl       VARCHAR(255) NULL,
    Status          VARCHAR(20) NOT NULL DEFAULT 'active'
                    CONSTRAINT CK_Users_Status CHECK (Status IN ('active','inactive','suspended','pending')), -- RB-1
    LastLoginAt     DATETIME2 NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    UpdatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 2. UserProfiles (thuc the yeu, quan he nhan dang 1:1 voi Users)
CREATE TABLE UserProfiles (
    UserId          UNIQUEIDENTIFIER NOT NULL PRIMARY KEY
                    REFERENCES Users(UserId) ON DELETE CASCADE,
    DateOfBirth     DATE NULL,
    Gender          VARCHAR(10) NULL,
    Address         NVARCHAR(255) NULL,
    Bio             NVARCHAR(MAX) NULL,
    UpdatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 3. Roles
CREATE TABLE Roles (
    RoleId          UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    RoleName        NVARCHAR(50) NOT NULL UNIQUE,
    Description     NVARCHAR(255) NULL
);
GO
 
-- 4. Permissions
CREATE TABLE Permissions (
    PermissionId    UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    PermissionCode  VARCHAR(50) NOT NULL UNIQUE,
    Description     NVARCHAR(255) NULL
);
GO
 
-- 5. RolePermissions (n-n)
CREATE TABLE RolePermissions (
    RoleId          UNIQUEIDENTIFIER NOT NULL REFERENCES Roles(RoleId) ON DELETE CASCADE,
    PermissionId    UNIQUEIDENTIFIER NOT NULL REFERENCES Permissions(PermissionId) ON DELETE CASCADE,
    PRIMARY KEY (RoleId, PermissionId)
);
GO
 
-- 6. Families
CREATE TABLE Families (
    FamilyId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyName      NVARCHAR(150) NOT NULL,
    Origin          NVARCHAR(255) NULL,
    Description     NVARCHAR(MAX) NULL,
    CoverImageUrl   VARCHAR(255) NULL,
    CreatedBy       UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 7. UserRoles (n-n, chien luoc anh xa EER {overlap, incomplete} - Muc 2.5)
CREATE TABLE UserRoles (
    UserId          UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId) ON DELETE CASCADE,
    RoleId          UNIQUEIDENTIFIER NOT NULL REFERENCES Roles(RoleId) ON DELETE CASCADE,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    PRIMARY KEY (UserId, RoleId, FamilyId)
);
GO
 
-- 8. Branches (tu tham chieu)
CREATE TABLE Branches (
    BranchId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    ParentBranchId  UNIQUEIDENTIFIER NULL REFERENCES Branches(BranchId),
    BranchName      NVARCHAR(100) NOT NULL,
    Description     NVARCHAR(255) NULL
);
GO
 
-- 9. FamilyMembers
CREATE TABLE FamilyMembers (
    MemberId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    BranchId        UNIQUEIDENTIFIER NULL REFERENCES Branches(BranchId) ON DELETE NO ACTION, -- tranh multiple cascade paths (Muc 4.3)
    UserId          UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    FullName        NVARCHAR(100) NOT NULL,
    Gender          VARCHAR(10) NULL,
    DateOfBirth     DATE NULL,
    DateOfDeath     DATE NULL,
    IsAlive         BIT NOT NULL DEFAULT 1,
    BirthPlace      NVARCHAR(255) NULL,
    GenerationLevel INT NULL,
    Biography       NVARCHAR(MAX) NULL,
    AvatarUrl       VARCHAR(255) NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT CK_FamilyMembers_AliveDeath CHECK ( -- RB-3
        (IsAlive = 0 AND DateOfDeath IS NOT NULL) OR (IsAlive = 1 AND DateOfDeath IS NULL)
    )
);
GO
 
-- 10. ParentChildRelationships (anh xa moi ket hop N:2 bi gioi han - Muc 2.3)
CREATE TABLE ParentChildRelationships (
    RelationshipId  UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    ParentMemberId  UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId),
    ChildMemberId   UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId),
    RelationshipType VARCHAR(20) NOT NULL DEFAULT 'biological',
    CONSTRAINT CK_NotSelfParent CHECK (ParentMemberId <> ChildMemberId), -- RB-4
    CONSTRAINT UQ_ParentChild UNIQUE (ParentMemberId, ChildMemberId)
);
GO
-- Han che "toi da 2 cha/me sinh hoc" (RB-6) can trigger AFTER INSERT/UPDATE - xem Muc 3.3.2 va 5.3
 
-- 11. Marriages (thuc the ket hop cho moi ket hop M:N)
CREATE TABLE Marriages (
    MarriageId      UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    Spouse1Id       UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId),
    Spouse2Id       UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId),
    MarriageDate    DATE NULL,
    DivorceDate     DATE NULL,
    Status          VARCHAR(20) NOT NULL DEFAULT 'married',
    CONSTRAINT CK_NotSelfMarriage CHECK (Spouse1Id <> Spouse2Id) -- RB-9 (phan 1)
);
GO
-- Han che "toi da 1 hon nhan dang 'married'" (RB-9, phan 2) can trigger - xem Muc 3.3.2
 
-- 12. MemberVerifications (thuc the manh, phu thuoc ton tai - xem lai Muc 2.4)
CREATE TABLE MemberVerifications (
    VerificationId  UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    MemberId        UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId) ON DELETE CASCADE,
    RequestedBy     UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    Status          VARCHAR(20) NOT NULL DEFAULT 'pending',
    EvidenceUrl     VARCHAR(255) NULL,
    VerifiedBy      UNIQUEIDENTIFIER NULL REFERENCES Users(UserId), -- RB-7: kiem tra vai tro o tang trigger/ung dung
    VerifiedAt      DATETIME2 NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 13. Posts
CREATE TABLE Posts (
    PostId          UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    AuthorId        UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    PostType        VARCHAR(20) NOT NULL DEFAULT 'text',
    Content         NVARCHAR(MAX) NULL,
    Visibility      VARCHAR(20) NOT NULL DEFAULT 'family',
    IsPinned        BIT NOT NULL DEFAULT 0,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    UpdatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 14. Comments (tu tham chieu)
CREATE TABLE Comments (
    CommentId       UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    PostId          UNIQUEIDENTIFIER NOT NULL REFERENCES Posts(PostId) ON DELETE CASCADE,
    AuthorId        UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    ParentCommentId UNIQUEIDENTIFIER NULL REFERENCES Comments(CommentId), -- RB-8: kiem tra o trigger/ung dung
    Content         NVARCHAR(MAX) NOT NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 15. Reactions (khoa da hinh: TargetType/TargetId khong FK cung)
CREATE TABLE Reactions (
    ReactionId      UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    UserId          UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId) ON DELETE CASCADE,
    TargetType      VARCHAR(20) NOT NULL,
    TargetId        UNIQUEIDENTIFIER NOT NULL,
    ReactionType    VARCHAR(20) NOT NULL DEFAULT 'like',
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT UQ_Reactions UNIQUE (UserId, TargetType, TargetId)
);
GO
 
-- 16. MediaAssets (khoa da hinh: OwnerType/OwnerId khong FK cung)
CREATE TABLE MediaAssets (
    MediaId         UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    OwnerType       VARCHAR(20) NOT NULL,
    OwnerId         UNIQUEIDENTIFIER NOT NULL,
    MediaUrl        VARCHAR(255) NOT NULL,
    MediaType       VARCHAR(20) NOT NULL DEFAULT 'image',
    Caption         NVARCHAR(255) NULL,
    UploadedBy      UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    UploadedAt      DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 17. Events
CREATE TABLE Events (
    EventId         UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    CreatedBy       UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    Title           NVARCHAR(200) NOT NULL,
    Description     NVARCHAR(MAX) NULL,
    EventType       VARCHAR(20) NOT NULL DEFAULT 'reunion',
    Location        NVARCHAR(255) NULL,
    StartTime       DATETIME2 NOT NULL,
    EndTime         DATETIME2 NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT CK_Events_TimeOrder CHECK (EndTime IS NULL OR StartTime <= EndTime) -- RB-2
);
GO
 
-- 18. EventParticipants
CREATE TABLE EventParticipants (
    EventId         UNIQUEIDENTIFIER NOT NULL REFERENCES Events(EventId) ON DELETE CASCADE,
    MemberId        UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId) ON DELETE NO ACTION, -- tranh multiple cascade paths
    RsvpStatus      VARCHAR(20) NOT NULL DEFAULT 'invited'
                    CONSTRAINT CK_EventParticipants_Rsvp CHECK (RsvpStatus IN ('invited','accepted','declined','maybe')),
    RespondedAt     DATETIME2 NULL,
    PRIMARY KEY (EventId, MemberId)
    -- RB-5: MemberId phai cung dong ho voi EventId.FamilyId - kiem tra o trigger/ung dung
);
GO
 
-- 19. EventReminders (thuc the yeu - khoa chu + khoa cuc bo RemindAt)
CREATE TABLE EventReminders (
    EventId         UNIQUEIDENTIFIER NOT NULL REFERENCES Events(EventId) ON DELETE CASCADE,
    RemindAt        DATETIME2 NOT NULL,
    Channel         VARCHAR(20) NOT NULL DEFAULT 'push',
    IsSent          BIT NOT NULL DEFAULT 0,
    PRIMARY KEY (EventId, RemindAt)
);
GO
 
-- 20. ProfessionalProfiles
CREATE TABLE ProfessionalProfiles (
    ProfileId       UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    MemberId        UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId) ON DELETE CASCADE,
    Occupation      NVARCHAR(100) NULL,
    Company         NVARCHAR(150) NULL,
    Industry        NVARCHAR(100) NULL,
    Position        NVARCHAR(100) NULL,
    YearsExperience INT NULL,
    Location        NVARCHAR(255) NULL
);
GO
 
-- 21. EducationProfiles
CREATE TABLE EducationProfiles (
    EducationId     UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    MemberId        UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId) ON DELETE CASCADE,
    SchoolName      NVARCHAR(150) NOT NULL,
    Degree          NVARCHAR(50) NULL,
    FieldOfStudy    NVARCHAR(100) NULL,
    StartYear       INT NULL,
    EndYear         INT NULL
);
GO
 
-- 22. HistoricalDocuments
CREATE TABLE HistoricalDocuments (
    DocumentId      UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    Title           NVARCHAR(200) NOT NULL,
    Description     NVARCHAR(MAX) NULL,
    DocType         VARCHAR(20) NULL,
    FileUrl         VARCHAR(255) NOT NULL,
    UploadedBy      UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    UploadedAt      DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 23. FamilyStories
CREATE TABLE FamilyStories (
    StoryId         UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    RelatedMemberId UNIQUEIDENTIFIER NULL REFERENCES FamilyMembers(MemberId),
    AuthorId        UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    Title           NVARCHAR(200) NOT NULL,
    Content         NVARCHAR(MAX) NOT NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 24. OutstandingMembers
CREATE TABLE OutstandingMembers (
    AchievementId   UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    MemberId        UNIQUEIDENTIFIER NOT NULL REFERENCES FamilyMembers(MemberId) ON DELETE CASCADE,
    AchievementTitle NVARCHAR(200) NOT NULL,
    Description     NVARCHAR(MAX) NULL,
    AchievementYear INT NULL
);
GO
 
-- 25. PhotoAlbums
CREATE TABLE PhotoAlbums (
    AlbumId         UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    AlbumName       NVARCHAR(150) NOT NULL,
    Description     NVARCHAR(MAX) NULL,
    CreatedBy       UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 26. AiQueryLogs
CREATE TABLE AiQueryLogs (
    QueryId         UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    UserId          UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    QueryType       VARCHAR(30) NOT NULL,
    QueryText       NVARCHAR(MAX) NOT NULL,
    ResponseText    NVARCHAR(MAX) NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 27. AiEmbeddings (khoa da hinh EntityType/EntityId)
CREATE TABLE AiEmbeddings (
    EmbeddingId     UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    EntityType      VARCHAR(30) NOT NULL,
    EntityId        UNIQUEIDENTIFIER NOT NULL,
    EmbeddingVector NVARCHAR(MAX) NOT NULL, -- mang so dang JSON, chua co kieu VECTOR goc (Muc 4.2, 5.2)
    ModelVersion    VARCHAR(50) NOT NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT UQ_AiEmbeddings UNIQUE (EntityType, EntityId)
);
GO
 
-- 28. AiRecommendations (khoa da hinh TargetType/TargetId)
CREATE TABLE AiRecommendations (
    RecommendationId UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    UserId          UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId) ON DELETE CASCADE,
    TargetType      VARCHAR(30) NOT NULL,
    TargetId        UNIQUEIDENTIFIER NOT NULL,
    Score           DECIMAL(5,4) NOT NULL,
    Reason          NVARCHAR(255) NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 29. Reports
CREATE TABLE Reports (
    ReportId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    FamilyId        UNIQUEIDENTIFIER NOT NULL REFERENCES Families(FamilyId) ON DELETE CASCADE,
    ReportType      VARCHAR(30) NOT NULL,
    Parameters      NVARCHAR(MAX) NULL CONSTRAINT CK_Reports_Params_Json CHECK (Parameters IS NULL OR ISJSON(Parameters) = 1),
    FileUrl         VARCHAR(255) NULL,
    GeneratedBy     UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    GeneratedAt     DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 30. AuditLogs (khoa tu tang, tan suat ghi cao - khoa da hinh EntityType/EntityId)
CREATE TABLE AuditLogs (
    LogId           BIGINT NOT NULL IDENTITY(1,1) PRIMARY KEY,
    UserId          UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    Action          VARCHAR(30) NOT NULL,
    EntityType      VARCHAR(50) NOT NULL,
    EntityId        UNIQUEIDENTIFIER NULL,
    OldValue        NVARCHAR(MAX) NULL,
    NewValue        NVARCHAR(MAX) NULL,
    IpAddress       VARCHAR(45) NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 31. SystemConfigurations
CREATE TABLE SystemConfigurations (
    ConfigId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    ConfigKey       VARCHAR(100) NOT NULL UNIQUE,
    ConfigValue     NVARCHAR(MAX) NULL,
    Description     NVARCHAR(255) NULL,
    UpdatedBy       UNIQUEIDENTIFIER NULL REFERENCES Users(UserId),
    UpdatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 32. Backups
CREATE TABLE Backups (
    BackupId        UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    BackupType      VARCHAR(20) NOT NULL,
    FilePath        VARCHAR(255) NOT NULL,
    Status          VARCHAR(20) NOT NULL DEFAULT 'in_progress',
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
 
-- 33. ModerationLogs (khoa da hinh ContentType/ContentId)
CREATE TABLE ModerationLogs (
    ModerationId    UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    ContentType     VARCHAR(20) NOT NULL,
    ContentId       UNIQUEIDENTIFIER NOT NULL,
    ModeratorId     UNIQUEIDENTIFIER NOT NULL REFERENCES Users(UserId),
    Action          VARCHAR(30) NOT NULL,
    Reason          NVARCHAR(255) NULL,
    CreatedAt       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);
GO
